-e ./core/engine

# what every block of the model puts out when it reads a transcript
-e ./core/tools/capture

# the two readers of a capture: the words at each depth, and the emotions
-e ./core/tools/jlens
-e ./core/tools/emotion

# readings for the page: runs the three tools on each transcript and serves the result
-e ./core/server
